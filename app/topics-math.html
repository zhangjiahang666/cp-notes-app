一、模运算

运算公式注意
加法(a+b) % m = (a%m + b%m) % m直接取模即可
减法(a-b) % m结果可能是负数，先加 m 再取模
乘法(a×b) % m先分别取模再相乘；大数相乘可能爆 long long
除法a/b 对模意义不能直接取模，要用「乘法逆元」
负数取模写法：((a % m) + m) % m；乘法取模溢出时用 __int128 或快速乘。

二、质数判定

只需枚举到 √n：若 n 有大于 √n 的因子，必有小于 √n 的因子。单次判定 O(√n)。

#include <bits/stdc++.h>
using namespace std;

bool is_prime(long long x) {
  if (x < 2) return false;
  for (long long i = 2; i * i <= x; i++)
    if (x % i == 0) return false;
  return true;
}

int main() {
  int n; cin >> n;
  cout << (is_prime(n) ? "是质数" : "不是质数") << "\n";
  return 0;
}
注意 i*i <= x 可能溢出 int，x 用 long long 且 i 同步用 long long。

三、筛法（批量求质数）

方法时间特点
埃氏筛O(n log log n)把每个质数的倍数标成合数；实现简单，n≤10⁷ 可用
欧拉筛O(n)每个合数只被最小质因子筛一次；顺带能记录最小质因子
区间筛O(R + (R-L+1) log log R)R 很大（如 10¹²）但区间 R-L 小（≤10⁶）时用
 
#include <bits/stdc++.h>
using namespace std;
const int N = 1000005;
bool vis[N];          // vis[i]=true 表示 i 是合数
int p[N], cnt;        // p[]: 筛出的质数表

void euler(int n) {   // 欧拉筛：每个合数只被最小质因子筛一次
  vis[0] = vis[1] = true;
  for (int i = 2; i <= n; i++) {
    if (!vis[i]) p[++cnt] = i;
    for (int j = 1; j <= cnt && i * p[j] <= n; j++) {
      vis[i * p[j]] = true;
      if (i % p[j] == 0) break;   // 保证只由最小质因子筛掉
    }
  }
}

int main() {
  int n; cin >> n;
  euler(n);
  for (int i = 1; i <= cnt; i++) cout << p[i] << " ";
  return 0;
}

埃氏筛更短：外层 for i=2..n 若 !vis[i] 则内层 for j=i*i;j<=n;j+=i 标合数即可。质数分布：n 以内约有 n/ln n 个质数。

四、质因子分解

唯一分解定理：任意正整数 n 可唯一写成 n = p₁^a₁ × p₂^a₂ × … × pₖ^aₖ。枚举到 √n 试除，最后剩余的 n>1 必是大质因子。

#include <bits/stdc++.h>
using namespace std;

void factor(int n) {
  for (int i = 2; i * i <= n; i++) {
    if (n % i == 0) {
      int c = 0;
      while (n % i == 0) { n /= i; c++; }
      cout << i << "^" << c << "  ";
    }
  }
  if (n != 1) cout << n << "^1";   // 剩下的大质因子
  cout << "\n";
}

int main() {
  int n; cin >> n;
  factor(n);
  return 0;
}
约数个数定理：d(n) = (a₁+1)(a₂+1)…(aₖ+1)（每个质因子选 0~aᵢ 次）。约数和定理：s(n)=∏(pᵢ^(aᵢ+1)−1)/(pᵢ−1)。

海量分解（10⁶ 个数）技巧：先用欧拉筛预处理每个数的最小质因子 f[i]，然后不断 n /= f[n]，单次分解 O(log n)。

五、gcd 与 lcm

#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

ll gcd(ll a, ll b) { return b ? gcd(b, a % b) : a; }  // 辗转相除
ll lcm(ll a, ll b) { return a / gcd(a, b) * b; }      // 先除后乘防溢出

int main() {
  ll a, b; cin >> a >> b;
  cout << gcd(a, b) << " " << lcm(a, b) << "\n";
  return 0;
}

C++17 可直接用 std::gcd / std::lcm（<numeric>）。

六、快速幂

把指数拆成二进制，底数反复平方，O(log n) 求 a^b % m。

#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

ll qpow(ll a, ll b, ll m) {
  ll r = 1 % m;
  while (b) {
    if (b & 1) r = r * a % m;
    a = a * a % m;
    b >>= 1;
  }
  return r;
}

int main() {
  ll a, b, m; cin >> a >> b >> m;
  cout << qpow(a, b, m) << "\n";
  return 0;
}

七、排列组合

公式写法
排列P(n,k) = n! / (n−k)!
组合C(n,k) = n! / (k! (n−k)!)
递推（杨辉三角）C(n,k) = C(n−1,k−1) + C(n−1,k)
 
#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const ll MOD = 1e9 + 7;
const int N = 100005;
ll fac[N], inv[N];

ll qpow(ll a, ll b) {
  ll r = 1;
  while (b) { if (b & 1) r = r * a % MOD; a = a * a % MOD; b >>= 1; }
  return r;
}

void pre() {                    // 预处理阶乘与阶乘逆元
  fac[0] = 1;
  for (int i = 1; i < N; i++) fac[i] = fac[i - 1] * i % MOD;
  inv[N - 1] = qpow(fac[N - 1], MOD - 2);   // 费马小定理求逆元
  for (int i = N - 2; i >= 0; i--) inv[i] = inv[i + 1] * (i + 1) % MOD;
}

ll C(int n, int k) {
  if (k < 0 || k > n) return 0;
  return fac[n] * inv[k] % MOD * inv[n - k] % MOD;
}

八、二项式定理

(a+b)ⁿ = Σ C(n,k)·a^k·b^(n−k)。由它可得：ΣC(n,k)=2ⁿ；奇偶项和相等。

九、exgcd 与乘法逆元

逆元：a·x ≡ 1 (mod m)，x 记作 a⁻¹；除法取模 = 乘逆元。费马小定理：m 为质数时 a⁻¹ = a^(m−2) mod m（即上面的快速幂）。

#include <bits/stdc++.h>
using namespace std;
typedef long long ll;

ll exgcd(ll a, ll b, ll &x, ll &y) {   // 解 ax+by=gcd(a,b)
  if (!b) { x = 1; y = 0; return a; }
  ll g = exgcd(b, a % b, y, x);
  y -= a / b * x;
  return g;
}

ll inv_mod(ll a, ll m) {   // 求逆元：m 不要求是质数
  ll x, y; exgcd(a, m, x, y);
  return (x % m + m) % m;
}

int main() {
  ll a, m; cin >> a >> m;
  cout << inv_mod(a, m) << "\n";
  return 0;
}

十、鸽笼原理

n+1 只鸽子放进 n 个笼子，至少一个笼子有 ≥2 只。常用：抽屉原理证存在性、求余数分类（mod n 下必有重复余数）。

十一、线性代数（向量与矩阵）

向量点积 a·b = Σ aᵢbᵢ；矩阵乘法 C[i][j] = Σ A[i][k]·B[k][j]。矩阵快速幂可加速递推（如斐波那契数列 O(log n)）。

#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N = 105;
ll a[N][N], b[N][N], c[N][N];

int main() {
  int n, m, p; cin >> n >> m >> p;      // a: n*m, b: m*p
  for (int i = 1; i <= n; i++)
    for (int j = 1; j <= m; j++) cin >> a[i][j];
  for (int i = 1; i <= m; i++)
    for (int j = 1; j <= p; j++) cin >> b[i][j];
  for (int i = 1; i <= n; i++)
    for (int k = 1; k <= m; k++)
      for (int j = 1; j <= p; j++)
        c[i][j] += a[i][k] * b[k][j];   // 注意累加可能超 int
  for (int i = 1; i <= n; i++, cout << "\n")
    for (int j = 1; j <= p; j++) cout << c[i][j] << " ";
  return 0;
}
速记：判质数试除到 √n；批量质数上欧拉筛；分解质因数试除 + 最小质因子加速；gcd 辗转相除；除法取模找逆元（费马小定理/ exgcd）。
